exports.handler = async function (event) { const corsHeaders = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "Content-Type", "Access-Control-Allow-Methods": "POST, OPTIONS", "Content-Type": "application/json", "Cache-Control": "no-store" }; // CORS preflight if (event.httpMethod === "OPTIONS") { return { statusCode: 204, headers: corsHeaders, body: "" }; } // POST only if (event.httpMethod !== "POST") { return { statusCode: 405, headers: corsHeaders, body: JSON.stringify({ error: "Method Not Allowed" }) }; } try { const apiKey = process.env.ANTHROPIC_API_KEY; if (!apiKey) { return { statusCode: 500, headers: corsHeaders, body: JSON.stringify({ error: "ANTHROPIC_API_KEY is not configured." }) }; } let body; try { body = JSON.parse(event.body || "{}"); } catch (e) { return { statusCode: 400, headers: corsHeaders, body: JSON.stringify({ error: "Invalid JSON." }) }; } const targetPhrase = typeof body.targetPhrase === "string" ? body.targetPhrase.trim().slice(0, 180) : ""; const recognisedText = typeof body.recognisedText === "string" ? body.recognisedText.trim().slice(0, 300) : ""; if (!targetPhrase || !recognisedText) { return { statusCode: 400, headers: corsHeaders, body: JSON.stringify({ error: "targetPhrase and recognisedText are required." }) }; } const prompt = ` You are providing very short speaking feedback for an adult English learner at A1-A2 level. IMPORTANT: You are NOT listening to audio. You are NOT judging pronunciation or accent. You are only comparing the words produced by speech recognition with the target phrase. Target phrase: "${targetPhrase}" Words recognised by the browser: "${recognisedText}" Use British English. Return ONLY valid JSON in exactly this structure: { "status": "match", "message": "Short supportive message.", "focus": "" } Allowed status values: - "match" - "try_again" - "unclear" RULES: 1. Use "match" when the recognised words clearly correspond to the target phrase. 2. Use "try_again" when there is a meaningful word difference and the learner should try the phrase again. 3. Use "unclear" when the recognition result is too incomplete or unreliable to make a useful comparison. 4. Never give: - a score - a percentage - a mark - a CEFR level - an accent judgement - a pronunciation grade 5. The message must: - be supportive - be suitable for an adult learner - be no longer than 160 characters 6. "focus" is optional. If used, it MUST be an exact continuous fragment copied from the target phrase. Maximum 60 characters. 7. If no useful focus is needed, return: "focus": "" Return JSON only. `; const response = await fetch( "https://api.anthropic.com/v1/messages", { method: "POST", headers: { "content-type": "application/json", "x-api-key": apiKey, "anthropic-version": "2023-06-01" }, body: JSON.stringify({ model: "claude-haiku-4-5-20251001", max_tokens: 220, temperature: 0, messages: [ { role: "user", content: prompt } ] }) } ); const data = await response.json(); if (!response.ok) { console.error( "Anthropic error:", response.status, data ); return { statusCode: 502, headers: corsHeaders, body: JSON.stringify({ error: "AI service unavailable." }) }; } const text = data && data.content && data.content[0] && data.content[0].text ? data.content[0].text.trim() : ""; let feedback; try { const cleaned = text .replace(/^```json\s*/i, "") .replace(/^```\s*/i, "") .replace(/\s*```$/i, "") .trim(); feedback = JSON.parse(cleaned); } catch (e) { console.error( "Invalid AI JSON:", text ); return { statusCode: 502, headers: corsHeaders, body: JSON.stringify({ error: "Invalid AI response." }) }; } const allowedStatuses = [ "match", "try_again", "unclear" ]; if (!allowedStatuses.includes(feedback.status)) { return { statusCode: 502, headers: corsHeaders, body: JSON.stringify({ error: "Invalid AI response." }) }; } let message = typeof feedback.message === "string" ? feedback.message.trim().slice(0, 160) : ""; let focus = typeof feedback.focus === "string" ? feedback.focus.trim().slice(0, 60) : ""; // Security / consistency: // focus must appear exactly inside targetPhrase. if ( focus && !targetPhrase .toLowerCase() .includes(focus.toLowerCase()) ) { focus = ""; } if (!message) { if (feedback.status === "match") { message = "Your words matched the model phrase."; } else if (feedback.status === "try_again") { message = "Try the phrase once more."; } else { message = "The words were not clear enough to compare."; } } return { statusCode: 200, headers: corsHeaders, body: JSON.stringify({ status: feedback.status, message: message, focus: focus }) }; } catch (error) { console.error( "pronunciation-feedback error:", error ); return { statusCode: 500, headers: corsHeaders, body: JSON.stringify({ error: "Server error." }) }; } };
